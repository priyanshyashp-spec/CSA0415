#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main()
{
    int source, destination;
    char buffer[100];
    int n;

    source = open("source.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);

    if (source == -1)
    {
        printf("Error creating source file\n");
        return 1;
    }

    write(source, "Hello World!\nThis is a file copy program.", 40);
    close(source);

    source = open("source.txt", O_RDONLY);

    if (source == -1)
    {
        printf("Error opening source file\n");
        return 1;
    }

    destination = open("destination.txt",
                       O_WRONLY | O_CREAT | O_TRUNC, 0644);

    if (destination == -1)
    {
        printf("Error creating destination file\n");
        close(source);
        return 1;
    }

    while ((n = read(source, buffer, sizeof(buffer))) > 0)
    {
        write(destination, buffer, n);
    }

    close(source);
    close(destination);

    printf("File copied successfully!\n");

    return 0;
}
